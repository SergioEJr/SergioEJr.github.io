% Derivative of x^3 as an exploded cube: a solid x^3 cube grows by dx on its
% three visible faces -> three x^2 dx slabs (the derivative 3x^2) plus faint
% subleading pieces that vanish in the limit. See _template.tex / _preamble.tex
% for the shared setup; this file keeps only the driver switch and the drawing.
\ifnum\pdfoutput>0
  \documentclass[tikz,border=3pt]{standalone}         % pdflatex → PDF proof
\else
  \documentclass[tikz,border=3pt,dvisvgm]{standalone} % latex   → web SVG
\fi
\input{_preamble}

\begin{document}
\begin{tikzpicture}[x=1mm, y=1mm]
  % Isometric-ish projection: cube-space (a=right, b=up, c=depth) -> page mm.
  % Depth axis goes up-and-right for a standard "3/4 view" cube.
  \def\ax{1}\def\ay{0}      % right  direction (page dx,dy per unit a)
  \def\bx{0}\def\by{1}      % up     direction
  \def\cx{0.5}\def\cy{0.35} % depth  direction (foreshortened)
  % Point macro: \Q{a}{b}{c} expands to a page coordinate in mm.
  \newcommand{\Q}[3]{({(#1)*\ax + (#2)*\bx + (#3)*\cx}, {(#1)*\ay + (#2)*\by + (#3)*\cy})}

  % Box macro: \boxABC{a0}{b0}{c0}{la}{lb}{lc}{face-style}{edge-style} draws the
  % three visible faces (top, front, right) of an axis-aligned box with near-
  % bottom-left corner (a0,b0,c0) and extents (la,lb,lc). Faces are filled with
  % {face-style} (for occlusion) and stroked with {edge-style}. Drawing filled
  % faces back-to-front (painter's order) makes nearer boxes occlude farther ones.
  \newcommand{\boxABC}[8]{%
    \def\aA{#1}\def\bB{#2}\def\cC{#3}%
    \def\la{#4}\def\lb{#5}\def\lc{#6}%
    % front face (c = c0), then top (b = b0+lb), then right (a = a0+la)
    \filldraw[#7, #8] \Q{\aA}{\bB}{\cC} -- \Q{\aA+\la}{\bB}{\cC} -- \Q{\aA+\la}{\bB+\lb}{\cC} -- \Q{\aA}{\bB+\lb}{\cC} -- cycle;
    \filldraw[#7, #8] \Q{\aA}{\bB+\lb}{\cC} -- \Q{\aA+\la}{\bB+\lb}{\cC} -- \Q{\aA+\la}{\bB+\lb}{\cC+\lc} -- \Q{\aA}{\bB+\lb}{\cC+\lc} -- cycle;
    \filldraw[#7, #8] \Q{\aA+\la}{\bB}{\cC} -- \Q{\aA+\la}{\bB+\lb}{\cC} -- \Q{\aA+\la}{\bB+\lb}{\cC+\lc} -- \Q{\aA+\la}{\bB}{\cC+\lc} -- cycle;
  }

  \def\X{40}   % big cube edge
  \def\dx{5}   % growth step

  % --- Left: solid cube x^3 (three visible faces). Origin at its near-bottom-left.
  \begin{scope}[shift={(0,0)}]
    % front face
    \draw[edge, ink] \Q{0}{0}{0} -- \Q{\X}{0}{0} -- \Q{\X}{\X}{0} -- \Q{0}{\X}{0} -- cycle;
    % top face
    \draw[edge, ink] \Q{0}{\X}{0} -- \Q{\X}{\X}{0} -- \Q{\X}{\X}{\X} -- \Q{0}{\X}{\X} -- cycle;
    % right face
    \draw[edge, ink] \Q{\X}{0}{0} -- \Q{\X}{\X}{0} -- \Q{\X}{\X}{\X} -- \Q{\X}{0}{\X} -- cycle;
    \node[text=ink] at \Q{\X/2}{\X/2}{0} {$x^3$};
  \end{scope}

  % --- Arrow between the two cubes.
  \def\Rx{95}  % page-mm x-shift to the right-hand exploded view
  \draw[edge, ink, ->] (66, {\X/2}) -- (\Rx-14, {\X/2});

  % --- Right: exploded (x+dx)^3. The x^3 core, with its 7 growth pieces pulled
  % out along their axes so every piece is visible: three x^2 dx slabs (colored,
  % the leading 3x^2) and three x dx^2 edges + one dx^3 corner (faint, subleading
  % — they vanish in the limit). Pieces are pulled by \pull along each growth
  % axis; colored faces are filled (translucent tint) so slabs read as solid.
  \begin{scope}[shift={(\Rx,0)}]
    \def\pull{9}
    % Per-axis pull offsets in page mm: +a is (\pull,0); +b is (0,\pull);
    % +c (depth) is (\pull*\cx, \pull*\cy).
    % Solid pieces: opaque page-color fill (figbg) so you can't see through them;
    % only the EDGES carry the piece's color.
    \tikzset{
      redface/.style    = {fill=figbg, draw=figred},
      blueface/.style   = {fill=figbg, draw=figblue},
      orangeface/.style = {fill=figbg, draw=figorange},
      coreface/.style   = {fill=figbg, draw=ink},
      subface/.style    = {fill=figbg, draw=muted, line width=1pt},
    }

    % The three growth directions all point toward the VIEWER's front: +a
    % (right), +b (up), and -c (out of the screen = down-left). Every piece is
    % pulled by \pull along the axes it grows on, so the subleading edges/corner
    % nestle in the FRONT gaps between the slabs.

    % --- x^3 core cube. ------------------------------------------------------
    \boxABC{0}{0}{0}{\X}{\X}{\X}{coreface}{edge}
    \node[text=ink] at \Q{\X/2}{\X/2}{0} {$x^3$};

    % --- Three leading x^2 dx slabs. -----------------------------------------
    % TOP slab (figred): grows +b, pulled up.
    \begin{scope}[shift={(0,\pull)}]
      \boxABC{0}{\X}{0}{\X}{\dx}{\X}{redface}{edge}
      % label on the (large) top face, centered.
      \node[text=figred] at \Q{\X/2}{\X+\dx}{\X/2} {$x^2\,dx$};
    \end{scope}
    % RIGHT slab (figblue): grows +a, pulled right.
    \begin{scope}[shift={(\pull,0)}]
      \boxABC{\X}{0}{0}{\dx}{\X}{\X}{blueface}{edge}
      % label on the (large) right face, centered, running vertically.
      \node[text=figblue, rotate=-90] at \Q{\X+\dx}{\X/2}{\X/2} {$x^2\,dx$};
    \end{scope}
    % FRONT slab (figorange): grows -c (toward viewer), the c=-dx..0 layer in
    % front of the core, pulled down-left so it sits clear in front.
    \begin{scope}[shift={({-\pull*\cx},{-\pull*\cy})}]
      \boxABC{0}{0}{-\dx}{\X}{\X}{\dx}{orangeface}{edge}
      \node[text=figorange] at \Q{\X/2}{\X/2}{-\dx} {$x^2\,dx$};
    \end{scope}

    % --- Subleading pieces: solid (figbg fill, faint muted solid edges),
    % nestled in the FRONT gaps. They vanish fastest in the limit.
    % x dx^2 edge (grows +b,-c): front-top edge, pulled up + down-left.
    \begin{scope}[shift={({-\pull*\cx},{\pull-\pull*\cy})}]
      \boxABC{0}{\X}{-\dx}{\X}{\dx}{\dx}{subface}{}
    \end{scope}
    % x dx^2 edge (grows +a,-c): front-right edge, pulled right + down-left.
    \begin{scope}[shift={({\pull-\pull*\cx},{-\pull*\cy})}]
      \boxABC{\X}{0}{-\dx}{\dx}{\X}{\dx}{subface}{}
    \end{scope}
    % x dx^2 edge (grows +a,+b): top-right edge, pulled right + up.
    \begin{scope}[shift={(\pull,\pull)}]
      \boxABC{\X}{\X}{0}{\dx}{\dx}{\X}{subface}{}
    \end{scope}
    % dx^3 corner (grows +a,+b,-c): front-top-right, pulled along all three.
    \begin{scope}[shift={({\pull-\pull*\cx},{\pull-\pull*\cy})}]
      \boxABC{\X}{\X}{-\dx}{\dx}{\dx}{\dx}{subface}{}
    \end{scope}
  \end{scope}
\end{tikzpicture}
\end{document}
