% Derivative of x^3 as an exploded cube: a solid x^3 cube grows by dx on its
% three visible faces -> three x^2 dx slabs (the derivative 3x^2) plus faint
% subleading pieces that vanish in the limit. See _template.tex / _preamble.tex
% for the shared setup; this file keeps only the driver switch and the drawing.
\ifnum\pdfoutput>0
  \documentclass[tikz,border=3pt]{standalone}         % pdflatex → PDF proof
\else
  \documentclass[tikz,border=3pt,dvisvgm]{standalone} % latex   → web SVG
\fi
\input{_preamble}

\begin{document}
\begin{tikzpicture}[x=1mm, y=1mm]
  % Isometric-ish projection: cube-space (a=right, b=up, c=depth) -> page mm.
  % Depth axis goes up-and-right for a standard "3/4 view" cube.
  \def\ax{1}\def\ay{0}      % right  direction (page dx,dy per unit a)
  \def\bx{0}\def\by{1}      % up     direction
  \def\cx{0.5}\def\cy{0.35} % depth  direction (foreshortened)
  % Point macro: \Q{a}{b}{c} expands to a page coordinate in mm.
  \newcommand{\Q}[3]{({(#1)*\ax + (#2)*\bx + (#3)*\cx}, {(#1)*\ay + (#2)*\by + (#3)*\cy})}

  \def\X{40}   % big cube edge
  \def\dx{10}  % growth step

  % --- Left: solid cube x^3 (three visible faces). Origin at its near-bottom-left.
  \begin{scope}[shift={(0,0)}]
    % front face
    \draw[edge, ink] \Q{0}{0}{0} -- \Q{\X}{0}{0} -- \Q{\X}{\X}{0} -- \Q{0}{\X}{0} -- cycle;
    % top face
    \draw[edge, ink] \Q{0}{\X}{0} -- \Q{\X}{\X}{0} -- \Q{\X}{\X}{\X} -- \Q{0}{\X}{\X} -- cycle;
    % right face
    \draw[edge, ink] \Q{\X}{0}{0} -- \Q{\X}{\X}{0} -- \Q{\X}{\X}{\X} -- \Q{\X}{0}{\X} -- cycle;
    \node[text=ink] at \Q{\X/2}{\X/2}{0} {$x^3$};
  \end{scope}

  % --- Arrow between the two cubes.
  \def\Rx{95}  % page-mm x-shift to the right-hand exploded view
  \draw[edge, ink, ->] (66, {\X/2}) -- (\Rx-6, {\X/2});

  % --- Right: exploded grown cube. Ghost of (x+dx)^3 as faint dashed edges.
  \begin{scope}[shift={(\Rx,0)}]
    \def\Xg{50} % \X + \dx
    % ghost: three visible faces of the grown cube as faint dashed edges
    \draw[hair, muted, dashed] \Q{0}{0}{0} -- \Q{\Xg}{0}{0} -- \Q{\Xg}{\Xg}{0} -- \Q{0}{\Xg}{0} -- cycle;
    \draw[hair, muted, dashed] \Q{0}{\Xg}{0} -- \Q{\Xg}{\Xg}{0} -- \Q{\Xg}{\Xg}{\Xg} -- \Q{0}{\Xg}{\Xg} -- cycle;
    \draw[hair, muted, dashed] \Q{\Xg}{0}{0} -- \Q{\Xg}{\Xg}{0} -- \Q{\Xg}{\Xg}{\Xg} -- \Q{\Xg}{0}{\Xg} -- cycle;

    % --- Three leading x^2 dx slabs, each pulled \pull off the ghost along its
    % outward normal so the "explosion" reads. Colors match the reference sketch.
    \def\pull{6}

    % TOP slab (figred): the x-face extruded up by dx (+b). Pulled up by \pull.
    \begin{scope}[shift={(0,\pull)}]
      \draw[edge, figred] \Q{0}{\X}{0} -- \Q{\X}{\X}{0} -- \Q{\X}{\X+\dx}{0} -- \Q{0}{\X+\dx}{0} -- cycle;                   % front
      \draw[edge, figred] \Q{0}{\X+\dx}{0} -- \Q{\X}{\X+\dx}{0} -- \Q{\X}{\X+\dx}{\X} -- \Q{0}{\X+\dx}{\X} -- cycle;        % top
      \draw[edge, figred] \Q{\X}{\X}{0} -- \Q{\X}{\X+\dx}{0} -- \Q{\X}{\X+\dx}{\X} -- \Q{\X}{\X}{\X} -- cycle;              % right
      \node[text=figred] at \Q{\X/2}{\X+\dx/2}{0} {$x^2\,dx$};
    \end{scope}

    % RIGHT slab (figblue): the x-face extruded right by dx (+a). Pulled right by \pull.
    \begin{scope}[shift={(\pull,0)}]
      \draw[edge, figblue] \Q{\X}{0}{0} -- \Q{\X+\dx}{0}{0} -- \Q{\X+\dx}{\X}{0} -- \Q{\X}{\X}{0} -- cycle;                 % front
      \draw[edge, figblue] \Q{\X}{\X}{0} -- \Q{\X+\dx}{\X}{0} -- \Q{\X+\dx}{\X}{\X} -- \Q{\X}{\X}{\X} -- cycle;             % top
      \draw[edge, figblue] \Q{\X+\dx}{0}{0} -- \Q{\X+\dx}{\X}{0} -- \Q{\X+\dx}{\X}{\X} -- \Q{\X+\dx}{0}{\X} -- cycle;       % right
      \node[text=figblue, rotate=-90] at \Q{\X+\dx}{\X/2}{\X/2} {$x^2\,dx$};
    \end{scope}

    % FRONT slab (figorange): the x-face extruded toward viewer by dx (-c depth).
    % Pulled toward viewer (-c) by \pull.
    \begin{scope}[shift={({-\pull*\cx},{-\pull*\cy})}]
      \draw[edge, figorange] \Q{0}{0}{0} -- \Q{\X}{0}{0} -- \Q{\X}{\X}{0} -- \Q{0}{\X}{0} -- cycle;                        % front face
      \draw[edge, figorange] \Q{0}{\X}{-\dx} -- \Q{\X}{\X}{-\dx} -- \Q{\X}{\X}{0} -- \Q{0}{\X}{0} -- cycle;                % top
      \draw[edge, figorange] \Q{\X}{0}{-\dx} -- \Q{\X}{\X}{-\dx} -- \Q{\X}{\X}{0} -- \Q{\X}{0}{0} -- cycle;                % right
      \node[text=figorange] at \Q{\X/2}{\X/2}{0} {$x^2\,dx$};
    \end{scope}

    % Subleading dx^3 corner cube at the far top-right-back of the ghost — the
    % tiny piece where all three slabs would overlap. Faint (muted dashed); it
    % vanishes fastest in the limit. The three x dx^2 columns are left implicit
    % in the ghost outline to keep the figure readable.
    \draw[hair, muted, dashed] \Q{\X}{\X}{\X} -- \Q{\Xg}{\X}{\X} -- \Q{\Xg}{\Xg}{\X} -- \Q{\X}{\Xg}{\X} -- cycle;                 % front
    \draw[hair, muted, dashed] \Q{\X}{\Xg}{\X} -- \Q{\Xg}{\Xg}{\X} -- \Q{\Xg}{\Xg}{\Xg} -- \Q{\X}{\Xg}{\Xg} -- cycle;            % top
    \draw[hair, muted, dashed] \Q{\Xg}{\X}{\X} -- \Q{\Xg}{\Xg}{\X} -- \Q{\Xg}{\Xg}{\Xg} -- \Q{\Xg}{\X}{\Xg} -- cycle;            % right
  \end{scope}
\end{tikzpicture}
\end{document}
